# BME280 integration thru SPI

I thought the last project took too long. This one took MUCH longer, mostly because I couldn't find a ready program code to paste in and run like I did till now, meaning I now had to write the SPI read and write functions on my own. <!-- more --> 

Of course I could just get the answer from ChatGPT, but I avoided looking up the full solution and insisted on looking up part by part for understanding various concepts in embedded C and ESP-IDF. I got a better understanding of writing to and reading from registers doing this project. 

I went through the ESP-IDF docs on the SPI master driver to understand usage of the SPI functions by IDF. I went through each type definitions (like `spi_bus_config_t`, `spi_device_interface_config_t` and others) to understand required parameters and such.

I also went through the Bosch BME280 datasheet to understand how the SPI connection works. This seemed to not be that much necessary (at this stage), since the BME280 driver code was written for both SPI and I2C, with the only major difference being the read and write functions supplied to the `bme280_t` struct. Nevertheless, I did understand a bit how different registers were set up in the sensor and their functions.

I also looked up the lcd example provided by esp idf.

#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#include "driver/spi_master.h"
#include "driver/gpio.h"
#include "esp_err.h"
#include "esp_log.h"
#include "hal/spi_types.h"

#include "bme280.h"

#define BME280_HOST SPI2_HOST

#define MOSI_PIN GPIO_NUM_23
#define MISO_PIN GPIO_NUM_19
#define SCK_PIN  GPIO_NUM_18
#define CS_PIN  GPIO_NUM_5

#define TAG "SPI"
#define TAG_BME280 "BME280"

static spi_device_handle_t bme280_handle;

void setup_spi(void)
{
    spi_bus_config_t bus_config = {
        .mosi_io_num = MOSI_PIN,
        .miso_io_num = MISO_PIN,
        .sclk_io_num = SCK_PIN,

        .quadwp_io_num = -1,
        .quadhd_io_num = -1,
    };

    esp_err_t ret = spi_bus_initialize(
        BME280_HOST,
        &bus_config,
        SPI_DMA_CH_AUTO
    );

    ESP_ERROR_CHECK(ret);
}

void register_device(void) 
{
    spi_device_interface_config_t devconfig = {
        .clock_speed_hz = 10 * 1000 * 1000,
        .mode = 0,
        .spics_io_num = CS_PIN,
        .queue_size = 1,
        .command_bits = 0,
        .address_bits = 0,
    };

    ESP_ERROR_CHECK(
        spi_bus_add_device(BME280_HOST, &devconfig, &bme280_handle)
    );
}

s8 BME280_SPI_bus_read(u8 dev_addr, u8 reg_addr, u8 *reg_data, u8 cnt) 
{
    s32 iError = BME280_INIT_VALUE;
    esp_err_t espRc;

    // const u8 cnt = 32;
    uint8_t tx[cnt + 1];
    uint8_t rx[cnt + 1];

    tx[0] = reg_addr | (1 << 7);
    memset(&tx[1], 0, cnt);

    spi_transaction_t transaction = {
        .length = (cnt + 1) * 8,
        .tx_buffer = tx,
        .rx_buffer = rx
    };

    espRc = spi_device_transmit(bme280_handle, &transaction);
    memcpy(reg_data, &rx[1], cnt);

    espRc == ESP_OK ? (iError = SUCCESS) : (iError = ERROR);

    return (s8)iError;
}

s8 BME280_SPI_bus_write(u8 dev_addr, u8 reg_addr, u8 *reg_data, u8 cnt) 
{
    s32 iError = BME280_INIT_VALUE;
    esp_err_t espRc;

    // if (cnt > 32) {
    //     return -1;
    // }

    // const u8 MAX_CNT = 32;
    uint8_t tx[cnt + 1];

    tx[0] = reg_addr & ~(1 << 7);
    memset(&tx[1], 0, cnt);

    memcpy(&tx[1], reg_data, cnt);

    spi_transaction_t transaction = {
        .length = (cnt + 1) * 8,
        .tx_buffer = tx,
        .rx_buffer = NULL
    };

    espRc = spi_device_transmit(bme280_handle, &transaction);

    espRc == ESP_OK ? (iError = SUCCESS) : (iError = ERROR);

    return (s8)iError;
}

void BME280_delay_msec(u32 msec)
{
	vTaskDelay(msec/portTICK_PERIOD_MS);
}

void bme280_reader_task(void *ignore)
{
	struct bme280_t bme280 = {
		.bus_write = BME280_SPI_bus_write,
		.bus_read = BME280_SPI_bus_read,
		.delay_msec = BME280_delay_msec
	};

	s32 com_rslt;
	s32 v_uncomp_pressure_s32;
	s32 v_uncomp_temperature_s32;
	s32 v_uncomp_humidity_s32;

	com_rslt = bme280_init(&bme280);

	com_rslt += bme280_set_oversamp_pressure(BME280_OVERSAMP_16X);
	com_rslt += bme280_set_oversamp_temperature(BME280_OVERSAMP_2X);
	com_rslt += bme280_set_oversamp_humidity(BME280_OVERSAMP_1X);

	com_rslt += bme280_set_standby_durn(BME280_STANDBY_TIME_1_MS);
	com_rslt += bme280_set_filter(BME280_FILTER_COEFF_16);

	com_rslt += bme280_set_power_mode(BME280_NORMAL_MODE);
	if (com_rslt == SUCCESS) {
		while(true) {
			vTaskDelay(1000/portTICK_PERIOD_MS);

			com_rslt = bme280_read_uncomp_pressure_temperature_humidity(
				&v_uncomp_pressure_s32, &v_uncomp_temperature_s32, &v_uncomp_humidity_s32);

			if (com_rslt == SUCCESS) {
                ESP_LOGI(TAG_BME280, "hello");
				ESP_LOGI(TAG_BME280, "%.2f degC / %.3f hPa / %.3f %%",
					bme280_compensate_temperature_double(v_uncomp_temperature_s32),
					bme280_compensate_pressure_double(v_uncomp_pressure_s32)/100, // Pa -> hPa
					bme280_compensate_humidity_double(v_uncomp_humidity_s32));
			} else {
				ESP_LOGE(TAG_BME280, "measure error. code: %d", com_rslt);
			}
		}
	} else {
		ESP_LOGE(TAG_BME280, "init or setting error. code: %d", com_rslt);
	}

	vTaskDelete(NULL);
}

void app_main(void)
{
    setup_spi();
    register_device();
    xTaskCreate(
        bme280_reader_task,
        "bme280_reader_task",
        2048,
        NULL,
        6,
        NULL
    );
}
